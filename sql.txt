SELECT T0.FY AS FY_N,T0.FM AS FM_N,T0.FW AS FW_N,T0.FM_LABEL,T0.FW_LABEL,T0.FW_START_DATE,T0.FW_END_DATE
       ,D1.REGION5_C,D1.PAIDCOMPSP,D1.FAMNON_C,D1.GROUP_C as LifeStage,D1.GroupSize,D1.repeaterIP,D1.GENDER,D1.newincomeIP_c--,D1.SHDLTrans_C,D1.agecat,D1.Channel_Group
	   ,CASE WHEN SUM(CASE WHEN T3.ATTN IS NULL THEN 0 ELSE T3.ATTN END) OVER(PARTITION BY T0.FY,T0.FW,D1.REGION5_C)=0 THEN 0 
	         ELSE CASE WHEN T33.ATTN IS NULL THEN 0 ELSE T33.ATTN END*CASE WHEN T3.ATTN IS NULL THEN 0 ELSE T3.ATTN END/SUM(CASE WHEN T3.ATTN IS NULL THEN 0 ELSE T3.ATTN END) OVER(PARTITION BY T0.FY,T0.FW,D1.REGION5_C) 
	    END AS ATTN
	   ,CASE WHEN T3.RESPID_CNT IS NULL THEN 0 ELSE T3.RESPID_CNT END AS SampleSize
FROM (SELECT DISTINCT FY,FQ,CASE WHEN FM>9 THEN FM-9 ELSE FM+3 END AS FM,FM_LABEL
             ,FW,FW_LABEL,FW_START_DATE,FW_END_DATE
	  FROM CIMATEMP.dbo.D_DATE 
      WHERE FY BETWEEN 2016 AND (SELECT MAX(FY) FROM CIMATEMP.dbo.D_DATE WHERE CAL_DATE<GETDATE())
	  ) T0
CROSS JOIN (
SELECT DISTINCT region5_C,paidcompsp,FAMNON_C,GROUP_C,GroupSize,repeaterIP,1 AS GENDER,newincomeIP_c
 FROM CIMATEMP.DBO.[SURVEY_VIZIP_CUST_CHANNEL] WHERE paidcompsp='PAID'
UNION
SELECT DISTINCT region5_C,paidcompsp,FAMNON_C,GROUP_C,GroupSize,repeaterIP,2,newincomeIP_c
 FROM CIMATEMP.DBO.[SURVEY_VIZIP_CUST_CHANNEL] WHERE paidcompsp='PAID'
) D1
LEFT JOIN (SELECT IP.FY,IP.FM,IP.FW,IP.REGION5_C,IP.paidcompsp,IP.FAMNON_C,IP.GROUP_C,IP.GroupSize,IP.repeaterIP,IP.GENDER,IP.newincomeIP_c
           ,SUM(IP.attendance) AS ATTN
		   ,COUNT(DISTINCT IP.RESPID) AS RESPID_CNT
           FROM (select respid,indicatorvalue AS GENDER
       ,FY,FM,FW,REGION5_C,paidcompsp,FAMNON_C,GROUP_C,GroupSize,repeaterIP_NEW AS repeaterIP,newincomeIP_c
,SUM(attendance/groupsize) as attendance
		   FROM (SELECT IP.*,CASE WHEN IP.DATE>='2017-04-02' THEN IP.repeaterIP ELSE OL.repeater END AS repeaterIP_NEW
from [CIMATEMP].[dbo].[SURVEY_VIZIP_CUST_CHANNEL] IP
LEFT JOIN [CIMATEMP].[dbo].SURVEY_VIZ_OL OL ON IP.RESPID=OL.respid_capi 
WHERE IP.paidcompsp='PAID') T
unpivot
(
  indicatorvalue
  for indicatorname in ([TRP_Gen_1]
      ,[TRP_Gen_2]
      ,[TRP_Gen_3]
      ,[TRP_Gen_4]
      ,[TRP_Gen_5]
      ,[TRP_Gen_6]
      ,[TRP_Gen_7]
      ,[TRP_Gen_8]
      ,[TRP_Gen_9]
      ,[TRP_Gen_10]
      ,[TRP_Gen_11]
      ,[TRP_Gen_12]
      ,[TRP_Gen_13]
      ,[TRP_Gen_14]
      ,[TRP_Gen_15]
      ,[TRP_Gen_16])
) unpiv
GROUP BY respid,indicatorvalue,FY,FM,FW,REGION5_C,paidcompsp,FAMNON_C,GROUP_C,GroupSize,repeaterIP_NEW,newincomeIP_c
) IP
		   LEFT JOIN [CIMATEMP].[dbo].SURVEY_VIZ_OL OL ON IP.RESPID=OL.respid_capi
		   WHERE IP.FY*100+IP.FW<=(SELECT MAX(FY*100+FW) FROM CIMATEMP.DBO.SURVEY_VIZ_IP)
           GROUP BY IP.FY,IP.FM,IP.FW,IP.REGION5_C,IP.paidcompsp,IP.FAMNON_C,IP.GROUP_C,IP.GroupSize,IP.repeaterIP,IP.GENDER,IP.newincomeIP_c
		   ) T3
ON T0.FY=T3.FY AND T0.FW=T3.FW
AND D1.region5_C=T3.region5_C AND D1.paidcompsp=T3.paidcompsp AND D1.FAMNON_C=T3.FAMNON_C 
AND D1.GROUP_C=T3.GROUP_C AND D1.GroupSize=T3.GroupSize AND D1.repeaterIP=T3.repeaterIP 
AND D1.GENDER=T3.GENDER AND D1.newincomeIP_c=T3.newincomeIP_c
LEFT JOIN (
SELECT FY,FW,region5_C,SUM(ATTENDANCE) AS ATTN FROM [CIMATEMP].[dbo].SURVEY_ORIGINS WHERE FY*100+FW<=(SELECT MAX(FY*100+FW) FROM CIMATEMP.DBO.SURVEY_VIZ_IP) AND paidcompsp='PAID' GROUP BY FY,FW,region5_C
) T33 ON T0.FY=T33.FY AND T0.FW=T33.FW AND D1.region5_C=T33.region5_C
